cmake_minimum_required(VERSION 3.24)
project(CredentialStorageTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Network Test DBus)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${REPO}/cmake/Warnings.cmake")
set(REMOTE "${REPO}/modules/remote/src")
add_library(credential_storage STATIC "${REMOTE}/cloud/CredentialStore.cpp"
    "${REMOTE}/cloud/CredentialBackend.cpp" "${REMOTE}/DaemonPaths.cpp"
    "${REMOTE}/cloud/SecretServiceCredentialBackend.cpp")
target_include_directories(credential_storage PUBLIC "${REMOTE}")
target_link_libraries(credential_storage PUBLIC Qt6::Core Qt6::Network Qt6::DBus)
if(APPLE)
    target_sources(credential_storage PRIVATE "${REMOTE}/cloud/KeychainCredentialBackend.cpp")
    target_link_libraries(credential_storage PRIVATE "-framework Security" "-framework CoreFoundation")
endif()
ketplus_enable_warnings(credential_storage)
add_executable(credential_store_tests ../CredentialStoreTest.cpp)
target_link_libraries(credential_store_tests PRIVATE credential_storage Qt6::Test)
ketplus_enable_warnings(credential_store_tests)
enable_testing()
add_test(NAME credential_store_tests COMMAND credential_store_tests)

set(CREDENTIAL_STORAGE_LIBRARY credential_storage)
set(CREDENTIAL_STORAGE_INCLUDES_SECRET_SERVICE TRUE)
include(NativeTests.cmake)
